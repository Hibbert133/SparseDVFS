cmake_minimum_required(VERSION 3.18)
project(sparsedvfs_thor_reference LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(fuse_dvfs SHARED cpp/FUSE.cpp)
target_compile_features(fuse_dvfs PRIVATE cxx_std_17)
target_compile_options(fuse_dvfs PRIVATE -O2 -Wall -Wextra -Wpedantic)
target_link_libraries(fuse_dvfs PRIVATE Threads::Threads)
option(FUSE_WITH_CUDA "Build explicit CUDA stream adapter" OFF)
if(FUSE_WITH_CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_library(fuse_cuda SHARED cpp/CUDAFUSE.cpp)
  target_compile_features(fuse_cuda PRIVATE cxx_std_17)
  target_compile_options(fuse_cuda PRIVATE -O2 -Wall -Wextra -Wpedantic)
  target_link_libraries(fuse_cuda PRIVATE fuse_dvfs CUDA::cudart Threads::Threads)
  set_target_properties(fuse_cuda PROPERTIES BUILD_RPATH "$ORIGIN")
endif()
